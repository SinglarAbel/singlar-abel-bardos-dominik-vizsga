CREATE DATABASE Eventra

CREATE TALBE felhasznalo(
f_az INT AUTO_INCREMENT,
nev VARCHAR(30),
felhasznaloNev VARCHAR(30) IS NOT NULL,
email VARCHAR(50) IS NOT NULL,
telefonSzam INT,
szuletesiEv INT,
CONSTRAINT pk_felhasz PRIMARY KEY(id)
)

CREATE TALBE programAdo(
p_az INT AUTO_INCREMENT PRIMARY KEY,
nev VARCHAR(100) IS NOT NULL,
email VARCHAR(100) IS NOT NULL,
szekhely VARCHAR(100),
CONSTRAINT pk_progado PRIMARY KEY(id)
)

CREATE TALBE esemeny(
e_az INT AUTO_INCREMENT PRIMARY KEY,
esemenyNev VARCHAR(100) IS NOT NULL,
telepules INT,
cim VARCHAR(100),
kor VARCHAR(20),
tipus int,
fizetos_e BOOLEAN,
stilus int,
informaciok int,
CONSTRAINT pk_esemeny PRIMARY KEY(id)
CONSTRAINT fk_telepules FOREIGN KEY(telepules) REFERENCES telepulesek(id),
CONSTRAINT fk_tipus FOREIGN KEY(tipus) REFERENCES tipusok(id),
CONSTRAINT fk_stilus FOREIGN KEY(stilus) REFERENCES stilusok(id),
CONSTRAINT fk_informaciok FOREIGN KEY(informaciok) REFERENCES informacio(id),
)

CREATE TABLE tipusok(
t_az INT AUTO_INCREMENT PRIMARY KEY,
megnevezés VARCHAR(40),
leiras VARChar(100)
)

CREATE TABLE helyszin (
id INT AUTO_INCREMENT PRIMARY KEY,
nev VARCHAR(100) NOT NULL,
szelesseg DECIMAL(10,7) NOT NULL,
hosszusag DECIMAL(10,7) NOT NULL
);

CREATE TABLE informacio(
i_az INT AUTO_INVREMENT PRIMARY KEY,
tartalom VARCHAR(200)
)

CREATE TABLE telepulesek (
t_az INT AUTO_INCREMENT PRIMARY KEY,
iranyitoszam VARCHAR(10) NOT NULL,
telepules VARCHAR(100) NOT NULL,
varmegye VARCHAR(100) NOT NULL
);
